# with extrusion (outer variables constrained inside a generalized let),
# nested lets, rows and records under extrusion.

fun f -> let g = fun x -> let h = fun y -> f x y in h in g

fun r -> let f = fun x -> r with {y = x} in {a = f 1; b = f true}
fun r -> let f = fun s -> if true then r else s in {a = f {x = 1}; b = f {y = 2}}
fun f -> let g = fun x -> f (f x) in g
let k = fun x -> fun y -> x in let c = k 1 in c true
fun z -> let a = fun x -> (z x).f in a

fun u -> let g = fun x -> u in let h = fun y -> g y y in h

fun q -> let f = fun x -> (q with {z = x}).z in f
fun f -> let g = fun x -> f {a = x} in let h = fun y -> g (g y) in h

fun f -> let id = fun x -> x in let g = fun y -> f (id y) in {a = g; b = id}
